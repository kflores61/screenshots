// A student passes when the average of two tests is at least 60.
// A first score of 0 ends the program, as required by this exercise.
start
   Declarations
      num firstTest
      num secondTest
      num average
      num PASSING = 60
   output "Enter first score or 0 to quit"
   input firstTest
   while firstTest <> 0
      output "Enter second score"
      input secondTest
      average = (firstTest + secondTest) / 2
      output "Average is ", average
      if average >= PASSING then
         output "Pass"
      else
         output "Fail"
      endif
      output "Enter first score or 0 to quit"
      input firstTest
   endwhile
stop
